dependency injection means supplying a function something with it needs, instead of making the function obtain itself
that “something” might be a database session, configuration, or an authenticated user.


from typing import Annotated
from fastapi import Depends, FastAPI, Header, HTTPException

app = FastAPI()

# Simplified demonstration; real apps verify tokens securely.
demo_users = {
    "demo-token": {"id": 1, "name": "Sanjay"}
}

def get_current_user(
    authorization: Annotated[str | None, Header()] = None,
):
    if authorization is None:
        raise HTTPException(status_code=401, detail="Missing token")

    scheme, _, token = authorization.partition(" ")
    user = demo_users.get(token)

    if scheme.lower() != "bearer" or user is None:
        raise HTTPException(status_code=401, detail="Invalid token")

    return user

@app.get("/profile")
def get_profile(user: Annotated[dict, Depends(get_current_user)]):
    return {"name": user["name"]}

@app.get("/dashboard")
def get_dashboard(user: Annotated[dict, Depends(get_current_user)]):
    return {"message": f"Welcome, {user['name']}"}


YIELD:
Use yield to provide the session, and finally block to close the session when the fastapi finishes using the dependency
For example, with a synchronous SQLAlchemy session factory named SessionLocal:

from typing import Annotated
from fastapi import Depends, FastAPI
from sqlalchemy.orm import Session

app = FastAPI()

def get_db():
    db = SessionLocal()  # Create a session for this request
    try:
        yield db        # Supply the session to the endpoint
    finally:
        db.close()      # Runs even if the endpoint raises an error

@app.get("/users/{user_id}")
def get_user(
    user_id: int,
    db: Annotated[Session, Depends(get_db)],
):
    user = db.get(User, user_id)
    return user

The lifecycle is:
Create session → yield session → endpoint uses it → close session
